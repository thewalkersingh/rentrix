-- Create new table
CREATE TABLE IF NOT EXISTS review_proofs (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    review_id BIGINT NOT NULL,
    storage_key VARCHAR(500) NOT NULL,
    original_filename VARCHAR(255),
    content_type VARCHAR(100) NOT NULL,
    size_bytes BIGINT NOT NULL,
    display_order INT NOT NULL DEFAULT 0,
    verified BOOLEAN NOT NULL DEFAULT FALSE,
    deleted BOOLEAN NOT NULL DEFAULT FALSE,
    created_at TIMESTAMP NOT NULL,
    updated_at TIMESTAMP NOT NULL,
    CONSTRAINT fk_review_proofs_review
        FOREIGN KEY (review_id) REFERENCES reviews(id)
        ON DELETE CASCADE
);

CREATE INDEX idx_review_proofs_review ON review_proofs (review_id);
CREATE INDEX idx_review_proofs_order ON review_proofs (review_id, display_order);
